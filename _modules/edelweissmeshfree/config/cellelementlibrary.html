def getCellElementClass(provider: str) -> type:
    """Get the class type of the requested CellElement provider.

    Parameters
    ----------
    provider
        The name of the cellelement provider to load.

    Returns
    -------
    type
        The cellelement provider class type.
    """

    if provider.lower() == "LagrangianMarmotCellElement".lower():
        from edelweissmeshfree.cellelements.marmotcellelement.lagrangianmarmotcellelement import (
            LagrangianMarmotCellElementWrapper,
        )

        return LagrangianMarmotCellElementWrapper
